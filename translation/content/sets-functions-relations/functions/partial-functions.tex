\olsection{جزوی فنکشن}

\begin{explain}
کدی کدی فنکشن دی تعریف نوں ڈھیلا کرنا فائدہ مند ہوندا اے، تاں جو
ہر ممکن انپٹ لئی فنکشن دا آؤٹ پٹ تعریف شدہ ہونا ضروری نہ ہووے۔
ایہو جہیاں نقشہ بندیاں نوں \emph{جزوی فنکشن} آکھیا جاندا اے۔
\end{explain}

\begin{defn}
اک \emph{جزوی فنکشن} $f \colon A \pto B$ ایہو جیہی نقشہ بندی اے
جیہڑی~$A$ دے ہر !!{element} نوں~$B$ دے ودھ توں ودھ اک !!{element}
نال جوڑدی اے۔ جے $f$،~$B$ دا اک عنصر $x \in A$ نال جوڑدا اے، تاں
اسی آکھدے آں کہ $f(x)$ \emph{تعریف شدہ} اے، تے نہیں تاں
\emph{غیر تعریف شدہ} اے۔ جے $f(x)$ تعریف شدہ ہووے، تاں اسی
$f(x) \fdefined$ لکھدے آں، نہیں تاں $f(x) \fundefined$۔ جزوی
فنکشن~$f$ دا \emph{دائرۂ تعریف}،~$A$ دا اوہ ذیلی سیٹ اے جتھے ایہ
تعریف شدہ اے، یعنی $\dom{f} = \Setabs{x \in A}{f(x) \fdefined}$۔
\end{defn}

\begin{ex}
ہر فنکشن $f\colon A \to B$ جزوی فنکشن وی اے۔ اوہ جزوی فنکشن
جیہڑے~$A$ اُتے ہر تھاں تعریف شدہ ہون---یعنی جنہاں نوں اسی ہُنے تک
بس فنکشن آکھدے آئے آں---\emph{کُل} فنکشن وی آکھلاندے نیں۔
\end{ex}

\begin{ex}
جزوی فنکشن $f \colon \Real \pto \Real$، جیہڑا $f(x) = 1/x$ نال
دتا اے، $x = 0$ اُتے غیر تعریف شدہ تے ہور ہر تھاں تعریف شدہ اے۔
\end{ex}

\begin{prob}
$f\colon A \pto B$ دتا ہووے، تاں جزوی فنکشن $g\colon B \pto A$
ایویں تعریف کرو: کسے وی $y \in B$ لئی، جے اکلوتا $x \in A$ اے
جیہدے لئی $f(x) = y$، تاں $g(y) = x$؛ نہیں تاں
$g(y) \fundefined$۔ وکھاؤ کہ جے $f$، اک بہ اک اے، تاں
$g(f(x)) = x$ ہر $x \in \dom{f}$ لئی، تے $f(g(y)) = y$ ہر
$y \in \ran{f}$ لئی۔
\end{prob}

\begin{defn}[جزوی فنکشن دا گراف]
من لو کہ $f\colon A \pto B$ اک جزوی فنکشن اے۔~$f$ دا
\emph{گراف} اوہ تعلق $R_f \subseteq A \times B$ اے جیہڑا ایویں
تعریف ہوندا اے:
\[
R_f = \Setabs{\tuple{x,y}}{f(x) = y}.
\]
\end{defn}

\begin{prop}
من لو کہ $R \subseteq A \times B$ وچ ایہ خاصیت اے کہ جدوں وی $Rxy$
تے $Rxy'$ ہون، تاں $y = y'$۔ فیر $R$ اوس جزوی فنکشن
$f\colon A \pto B$ دا گراف اے جیہڑا ایویں تعریف ہوندا اے: جے کوئی
$y$ اے جیہدے لئی $Rxy$، تاں $f(x) = y$، نہیں تاں
$f(x) \fundefined$۔ جے $R$ نالے \emph{سریئل} ہووے، یعنی ہر
$x \in A$ لئی کوئی $y \in B$ ہووے جیہدے لئی $Rxy$، تاں $f$ کُل اے۔
\end{prop}

\begin{proof}
من لو کہ کوئی $y$ اے جیہدے لئی $Rxy$۔ جے کوئی ہور $y' \neq y$
ہوندا جیہدے لئی $Rxy'$، تاں $R$ اُتے شرط دی خلاف ورزی ہوندی۔ سو
جے کوئی $y$ اے جیہدے لئی $Rxy$، تاں اوہ $y$ اکلوتا اے، تے ایس
لئی $f$ صحیح طور تے تعریف شدہ اے۔ ظاہر اے کہ $R_f = R$، تے $f$
کُل اے جے~$R$ سریئل ہووے۔
\end{proof}
