\olsection{بنیادی گلاں}

\begin{explain}
اک \emph{فنکشن} ایہو جیہی نقشہ بندی اے جیہڑی دتے ہوئے سیٹ دے ہر
!!{element} نوں کسے (ہور) دتے ہوئے سیٹ دے اک متعین !!{element} ول
گھلّدی اے۔ مثال لئی، $1$ جمع کرن دا عمل اک فنکشن تعریف کردا اے: ہر
عدد~$n$ نوں اک اکلوتے عدد~$n+1$ نال جوڑیا جاندا اے۔

ہور عام طور تے، فنکشن جوڑے، تِکے وغیرہ انپٹ دے طور تے لے سکدے نیں
تے کسے قسم دا آؤٹ پٹ دیندے نیں۔ کئی فنکشن سانوں بنیادی حساب توں
واقف نیں۔ مثال لئی، جمع تے ضرب فنکشن نیں۔ اوہ دو عدد لیندے نیں تے
تیجا عدد دیندے نیں۔

ایس ریاضیاتی، مجرد معنی وچ، فنکشن اک \emph{کالا ڈبّا} اے: اہم صرف
ایہ اے کہ کیہڑا آؤٹ پٹ کیہڑے انپٹ نال جوڑیا ہویا اے، نہ کہ آؤٹ پٹ
کڈھن دا طریقہ۔
\end{explain}

\begin{defn}[فنکشن]
اک \emph{فنکشن} $f \colon A \to B$، سیٹ~$A$ دے ہر !!{element} دی
سیٹ~$B$ دے اک !!{element} ول نقشہ بندی اے۔

اسی $A$ نوں فنکشن~$f$ دا \emph{دائرۂ تعریف} تے $B$ نوں فنکشن~$f$ دا
\emph{ہدف سیٹ} آکھدے آں۔ سیٹ~$A$ دے !!{element}s، فنکشن~$f$ دے انپٹ
یا \emph{دلیلاں} آکھلاندے نیں؛ تے سیٹ~$B$ دا اوہ !!{element} جیہنوں
دلیل~$x$ نال فنکشن~$f$ جوڑدا اے، \emph{فنکشن~$f$ دی قدر} آکھلاندا اے
جدوں دلیل~$x$ ہووے، تے اوہنوں~$f(x)$ لکھدے آں۔

\emph{حاصل قدراں دا سیٹ} $\ran{f}$، فنکشن~$f$ دے ہدف سیٹ دا اوہ ذیلی
سیٹ اے جیہدے وچ فنکشن~$f$ دیاں اوہ قدراں شامل نیں جیہڑیاں کسے دلیل
لئی حاصل ہوندیاں نیں؛ $\ran{f} = \Setabs{f(x)}{x \in A}$۔
\end{defn}

\olref{fig:function} والا خاکہ فنکشناں بارے سوچن وچ مدد دے سکدا اے۔
کھبے پاسے دا بیضہ نما دائرہ فنکشن دا \emph{دائرۂ تعریف} ظاہر کردا اے؛
سجے پاسے دا بیضہ نما دائرہ فنکشن دا \emph{ہدف سیٹ} ظاہر کردا اے؛ تے
تیر دائرۂ تعریف دی اک \emph{دلیل} توں ہدف سیٹ دی اوہدے نال جڑی
\emph{قدر} ول اشارہ کردا اے۔

\begin{ex}
ضرب قدرتی عدداں دے جوڑے انپٹ دے طور تے لیندی اے تے اوہناں نوں
قدرتی عدداں نال آؤٹ پٹ دے طور تے جوڑدی اے، سو ایہ $\Nat \times \Nat$
(دائرۂ تعریف) توں $\Nat$ (ہدف سیٹ) ول جاندی اے۔ حاصل قدراں دا سیٹ وی
$\Nat$ ای نکلدا اے، کیوں جے ہر $n \in \Nat$، حاصل ضرب $n \times 1$ اے۔
\end{ex}

\begin{ex}
ضرب اک فنکشن اے، کیوں جے ایہ ہر انپٹ---قدرتی عدداں دے ہر جوڑے---نوں
اکو آؤٹ پٹ نال جوڑدی اے: $\times \colon \Nat^2 \to \Nat$۔ ایس دے
اُلٹ، قدرتی عدد~$n$ نوں ایہو جہے حقیقی عدد~$x$ نال جوڑنا کہ $x^2 = n$
ہووے، فنکشن نہیں بناؤندا، کیوں جے ہر مثبت صحیح عدد~$n$ دے دو مربع
جذر نیں: $\sqrt{n}$ تے~$-\sqrt{n}$۔ صرف غیر منفی (اصل) مربع جذر واپس
دے کے اسی ایہنوں فنکشن بنا سکدے آں: $\sqrt{\phantom{X}} \colon \Nat \to \Real$۔
% Source correction OLFUN-002: English says "positive" although Nat includes 0.
\end{ex}

\begin{ex}
اوہ تعلق جیہڑا جماعت دے ہر طالب علم نوں اوہدے آخری گریڈ نال جوڑدا
اے، فنکشن اے---اکو جماعت وچ کسے طالب علم نوں دو وکھ وکھ آخری گریڈ
نہیں مل سکدے۔ اوہ تعلق جیہڑا جماعت دے ہر طالب علم نوں اوہدے ماں پیو
نال جوڑدا اے، فنکشن نہیں: طالب علماں دے صفر، یا دو، یا دو توں ودھ
ماں پیو ہو سکدے نیں۔
\end{ex}

\begin{explain}
اسی ہر ممکن دلیل لئی فنکشن دی قدر نوں کسے ٹھیک ٹھیک طریقے نال دس
کے فنکشن تعریف کر سکدے آں۔ ایہ کرن دے وکھ وکھ طریقے نیں: فارمولا
دینا، قدر کڈھن دا طریقہ بیان کرنا، یا ہر دلیل لئی قدر دی فہرست دینا۔
فنکشن نوں جِویں وی تعریف کریے، سانوں ایہ یقینی بناؤنا چاہیدا اے کہ
ہر دلیل لئی اک، تے صرف اک، قدر دسی گئی ہووے۔
\end{explain}


\begin{ex}
من لو کہ $f \colon \Nat \to \Nat$ ایویں تعریف کیتا گیا اے کہ
$f(x) = x+1$۔ ایہ تعریف $f$ نوں ایہو جہے فنکشن دے طور تے دسدی اے
جیہڑا قدرتی عدد لیندا اے تے قدرتی عدد آؤٹ پٹ دے طور تے دیندا اے۔
ایہ سانوں دسدی اے کہ قدرتی عدد~$x$ دتے جان اُتے، $f$ اوہدا
جانشین~$x+1$ دے گا۔ ایس صورت وچ، ہدف سیٹ $\Nat$، فنکشن~$f$ دا حاصل
قدراں دا سیٹ نہیں، کیوں جے قدرتی عدد~$0$ کسے وی قدرتی عدد دا جانشین
نہیں۔ فنکشن~$f$ دا حاصل قدراں دا سیٹ سارے مثبت صحیح عدداں دا سیٹ،
$\Int^{+}$، اے۔
\end{ex}

\begin{ex}\ollabel{examplefunext}
من لو کہ $g \colon \Nat \to \Nat$ ایویں تعریف کیتا گیا اے کہ
$g(x) = x+2-1$۔ ایہ سانوں دسدا اے کہ $g$ اک فنکشن اے جیہڑا قدرتی
عدد لیندا اے تے قدرتی عدد دیندا اے۔ قدرتی عدد~$x$ دتے جان اُتے، $g$
عدد~$x$ دے جانشین دے جانشین دا پیشرو، یعنی $x+1$، دے گا۔
% Source correction OLFUN-003: English switches from input n to x; target consistently uses x.
\end{ex}

\begin{explain}
اسی ہُنے دو فنکشناں، $f$ تے $g$، اُتے غور کیتا اے جنہاں دیاں
\emph{تعریفاں} وکھریاں نیں۔ پر ایہ \emph{اکو فنکشن} نیں۔ آخر، ہر
قدرتی عدد~$n$ لئی ساڈے کول $f(n) = n+1 = n+2-1 = g(n)$ اے۔ ہور
لفظاں وچ، $f$ تے~$g$ دیاں ساڈیاں تعریفاں وکھ وکھ مساواتاں راہیں
اکو نقشہ بندی دسدیاں نیں۔ سو اسی ضمنی طور تے فنکشناں لئی عنصراں نال
تعیّن دے اک اصول اُتے بھروسا کر رہے آں:
\[
  \text{جے }\forall x\, f(x) = g(x)\text{، تاں }f = g
\]
بشرطیکہ $f$ تے~$g$ دا دائرۂ تعریف تے ہدف سیٹ اکو ہون۔
\end{explain}

\begin{ex}
اسی صورتاں دے حساب نال وی فنکشن تعریف کر سکدے آں۔ مثال لئی، اسی
$h \colon \Nat \to \Nat$ نوں ایویں تعریف کر سکدے آں:
\[
h(x) =
\begin{cases}
  \frac{x}{2} & \text{جے $x$ جفت اے} \\
  \frac{x+1}{2} & \text{جے $x$ طاق اے۔}
\end{cases}
\]
کیوں جے ہر قدرتی عدد یا جفت اے یا طاق، ایس فنکشن دا آؤٹ پٹ ہمیشہ
قدرتی عدد ہووے گا۔ بس ایہ یاد رکھو کہ جے تسی صورتاں دے حساب نال
فنکشن تعریف کرو، تاں ہر ممکن انپٹ عین اک صورت وچ آؤنا چاہیدا اے۔
کجھ موقعیاں اُتے ایس لئی ثبوت دینا پئے گا کہ صورتاں سارے امکان
گھیردیاں نیں تے آپس وچ اکٹھیاں لاگو نہیں ہوندیاں۔
\end{ex}
