\olfileid{cmp}{thy}{tot}
\olsection{کُل ہون دا فیصلہ ممکن نہیں}

آؤ $s$-$m$-$n$ قضیے دی اک ہور
ورتوں ویکھئیے، جیہدے نال اسی
وکھاواں گے کہ اک سیٹ قابلِ
حساب نہیں۔ $\fn{Tot}$ کُل
قابلِ حساب فنکشناں دے
اشارییاں دا سیٹ اے، یعنی:
\[
\fn{Tot} = \Setabs{x}{\text{ہر $y$ لئی، $\cfind{x}(y)\fdefined$}}.
\]

\begin{prop}
\ollabel{prop:total}
$\fn{Tot}$ قابلِ حساب نہیں۔
\end{prop}

\begin{proof}
ایہ ویکھن لئی کہ $\fn{Tot}$ قابلِ حساب نہیں،
ایہ وکھانا کافی اے کہ $K$ دی ایس
ول تقلیل ہو سکدی اے۔ $h(x,y)$
نوں ایداں تعریف کرو:
\[
h(x,y) \simeq
\begin{cases}
0 & \text{جے $x \in K$} \\
\fundefined & \text{ہور صورت وچ}
\end{cases}
\]
دھیان دیو کہ $h(x,y)$
دا $y$ نال بالکل کوئی
واسطہ نہیں۔ ایہ ویکھنا اوکھا نہیں کہ $h$ جزوی
قابلِ حساب اے: انپٹ
$x, y$ اُتے $h$ دی قدر
کڈھن لئی پہلے $\cfind{x}$ دی انپٹ~$x$ اُتے
حسابی کارروائی دی مشینی نقل کرو؛ جے
ایہ حسابی کارروائی
رُک جائے، تاں $h(x,y)$
دا آؤٹ پٹ $0$ آؤندا اے
تے اوہ رُک جاندا اے۔
سو $h(x,y)$ عین
$\Zero(\umin{s}{T(x,x,s)})$
اے، جتھے $\Zero$
مستقل صفر فنکشن اے۔

$s$-$m$-$n$ قضیے
مطابق اک بنیادی بازگشتی
فنکشن~$k(x)$ موجود اے
ایداں کہ ہر $x$ تے~$y$
لئی:
\[
\cfind{k(x)}(y) =
\begin{cases}
0 & \text{جے $x \in K$} \\
\fundefined & \text{ہور صورت وچ}
\end{cases}
\]
ایس لئی $\cfind{k(x)}$
کُل اے جے $x \in K$
ہووے، تے نہیں تاں
ہر انپٹ اُتے غیر تعریف
شدہ اے۔ لہٰذا $k$،
$K$ دی~$\fn{Tot}$ ول
تقلیل اے۔
\end{proof}

\begin{digress}
اصل وچ $\fn{Tot}$
قابلِ حساب گنتی وار
فہرست بنن جوگ وی
نہیں؛ ایہدی پیچیدگی
«حسابی درجہ بندی»
وچ ہور اُتّے اے۔
پر ایتھے اسی ایس
مزید مضبوط نتیجے
دی بحث نہیں کراں گے۔
\end{digress}
