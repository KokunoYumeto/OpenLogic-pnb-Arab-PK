\olsection{ساختی قاعدے (Structural Rules)}

سانوں کجھ ایہو جیہے ساختی قاعدیاں (structural rules) دی وی لوڑ اے جیہڑے سیکوئنٹ دے کھبے تے
سجے پاسے !!{sentence}s نوں مُڑ ترتیب دین (rearrange) دی اجازت دیندے نیں۔ چونکہ
منطقی قاعدیاں (logical rules) دا تقاضا اے کہ مقدمے (premise) وچ اوہ !!{sentence}s، جِنہاں اُتے
قاعدہ لاگو ہُندا اے، یا تاں بالکل کھبے سرے یا بالکل سجے سرے (far left or far right) اُتے ہون،
ایس لئی سانوں اک ``تبادلے'' دے قاعدے (Exchange rule) دی لوڑ اے جیہڑا !!{sentence}s
نوں ٹھیک تھاں (required position) اُتے لے جان دی اجازت دیندا اے۔ کدی کدی دو اکّو جیہے
!!{sentence}s (two identical sentences) نوں اک وچ ملا سکنا، تے کسے وی پاسے اک !!a{sentence}
شامل کر سکنا (add on either side) وی ضروری اے۔

\subsection{کمزور کرنا (Weakening)}

\begin{defish}
\Axiom$ \Gamma \fCenter \Delta $
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$ !A, \Gamma \fCenter \Delta$
\DisplayProof
\hfill
\Axiom$ \Gamma \fCenter \Delta$
\RightLabel{\RightR{\Weakening}}
\UnaryInf$ \Gamma \fCenter \Delta, !A$
\DisplayProof
\end{defish}

\subsection{سکیڑنا (Contraction)}

\begin{defish}
\Axiom$ !A, !A, \Gamma \fCenter \Delta $
\RightLabel{\LeftR{\Contraction}}
\UnaryInf$ !A, \Gamma \fCenter \Delta$
\DisplayProof
\hfill
\Axiom$ \Gamma \fCenter \Delta, !A, !A$
\RightLabel{\RightR{\Contraction}}
\UnaryInf$ \Gamma \fCenter \Delta, !A$
\DisplayProof
\end{defish}

\subsection{تبادلہ (Exchange)}

\begin{defish}
\Axiom$ \Gamma, !A, !B, \Pi \fCenter \Delta $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$ \Gamma, !B, !A, \Pi \fCenter \Delta$
\DisplayProof
\hfill
\Axiom$ \Gamma \fCenter \Delta, !A, !B, \Lambda$
\RightLabel{\RightR{\Exchange}}
\UnaryInf$ \Gamma \fCenter \Delta, !B, !A, \Lambda$
\DisplayProof
\end{defish}

کمزور کرن، سکیڑن تے تبادلے دے کئی لگاتار استنباط (series of inferences) اکثر دُہریاں
استنباطی لیکاں (double inference lines) نال وکھائے جان گے۔

ہِٹھلا قاعدہ، جہنوں ``کٹ'' (Cut) آکھیا جاندا اے، سختی نال ویکھیا جاوے تاں
ضروری نہیں (not strictly necessary)، پر !!{derivation}s نوں مُڑ ورتن تے اکٹھیاں کرن (reuse and combine) نوں بہت
سَوکھا بنا دیندا اے۔
\begin{defish}
\[
\Axiom$ \Gamma \fCenter \Delta, !A$
\Axiom$ !A, \Pi \fCenter \Lambda $
\RightLabel{\Cut}
\BinaryInf$ \Gamma, \Pi \fCenter \Delta, \Lambda$
\DisplayProof
\]
\end{defish}
